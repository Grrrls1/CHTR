def read_csv(p):return list(csv.DictReader(p.open()))
def write_csv(p,rows):
    keys=list(dict.fromkeys(k for r in rows for k in r))
    with p.open('w') as f:
        w=csv.DictWriter(f,fieldnames=keys);w.writeheader();w.writerows(rows)
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def number(r,k):return float(r[k])
def fmt(r,k,d=4):return f"{float(r[k+'_mean']):.{d}f} ± {float(r[k+'_sd']):.{d}f}" if r[k+'_mean']!='' else 'undefined'

assert json.loads((P/'delivery_status.json').read_text())['state']=='complete'
assert json.loads((I/'acceptance.json').read_text())['state']=='passed'
for parent in [P,I]:
    for name,h in json.loads((parent/'artifact_manifest.json').read_text()).items():assert sha(parent/name)==h,(parent,name)
base=read_csv(P/'summary_mean_std.csv');image=read_csv(I/'summary_mean_std.csv')
by_image={(r['model_id'],r['test_domain']):r for r in image}
combined=[]
for r in base:
    r=dict(r);im=by_image[r['model_id'],r['test_domain']]
    for k in ['aiou_mean','aiou_sd']:assert abs(float(r[k])-float(im[k]))<1e-12
    assert abs(float(r['character_trajectory_iou_mean'])-float(im['original_gt_xyw_char_iou_mean']))<1e-12
    for k,v in im.items():
        if k.startswith('image_char_iou_') or k.startswith('gt_xyw_image_iou_'):r[k]=v
    r.update(image_char_iou_reference='input image Otsu foreground',
             ord_iou_reference='GT XYW raster; original unchanged',
             original_char_iou_reference='GT XYW raster; retained diagnostic')
    combined.append(r)
write_csv(O/'summary_mean_std.csv',combined)
by_rep={(r['model_id'],r['test_domain'],r['seed']):r for r in read_csv(I/'per_repeat_metrics.csv')}
repeat=[]
for r in read_csv(P/'per_repeat_metrics.csv'):
    r=dict(r);im=by_rep[r['model_id'],r['test_domain'],r['seed']]
    assert abs(float(r['aiou'])-float(im['aiou']))<1e-12
    assert abs(float(r['character_trajectory_iou'])-float(im['original_gt_xyw_char_iou']))<1e-12
    r['image_char_iou']=im['image_char_iou'];r['gt_xyw_image_iou']=im['gt_xyw_image_iou'];repeat.append(r)
write_csv(O/'per_repeat_metrics.csv',repeat)
by_pair={(r['test_domain'],r['seed'],r['model_a'],r['model_b']):r for r in read_csv(I/'paired_differences.csv')}
pairs=[]
for r in read_csv(P/'paired_differences.csv'):
    r=dict(r);im=by_pair[r['test_domain'],r['seed'],r['model_a'],r['model_b']]
    r['image_char_iou']=im['image_char_iou'];pairs.append(r)
write_csv(O/'paired_differences.csv',pairs)
by_ps={}
for r in pairs:by_ps.setdefault((r['test_domain'],r['model_a'],r['model_b']),[]).append(float(r['image_char_iou']))
paired_summary=read_csv(P/'paired_differences_mean_std.csv')
for r in paired_summary:
    vals=by_ps[r['test_domain'],r['model_a'],r['model_b']]
    r['image_char_iou_mean']=mean(vals);r['image_char_iou_sd']=stdev(vals);r['image_char_iou_defined_repeats']=len(vals)
write_csv(O/'paired_differences_mean_std.csv',paired_summary)
assert len(combined)==6 and len(repeat)==30 and len(pairs)==30
# Ensure consolidation copied OrdIoU exactly, rather than measuring a new definition.
for old,new in zip(base,combined):
    for k in old:
        if k.startswith('ordered_trajectory_iou_'):assert old[k]==new[k]
for r in combined:
    vals=[float(x['image_char_iou']) for x in repeat if (x['model_id'],x['test_domain'])==(r['model_id'],r['test_domain'])]
    assert abs(mean(vals)-float(r['image_char_iou_mean']))<1e-12
    assert abs(stdev(vals)-float(r['image_char_iou_sd']))<1e-12
